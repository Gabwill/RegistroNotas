PSEUDOCODIGO PARA REGISTRO DE NOTAS

// LA CLASE CREA VARIOS ESTUDIANTES, TODOS CON LOS MISMOS 
// ATRIBUTOS PERO CON VALORES DIFERENTES

CLASE Estudiante

    ATRIBUTOS:
        nombre       : Cadena
        idEstudiante : Cadena
        notas[5]     : Real
        promedio     : Real

    METODOS:

        // Constructor
        Estudiante(nombre, idEstudiante, notas[])
            this.nombre       ← nombre
            this.idEstudiante ← idEstudiante
            this.notas        ← notas
            this.promedio     ← 0
        FinConstructor

        // Calcula y guarda el promedio del estudiante
        CalcularPromedio(): Real
            Definir suma Como Real
            Definir i    Como Entero
            suma ← 0
            Para i ← 0 hasta 4 hacer
                suma ← suma + this.notas[i]
            FinPara
            this.promedio ← suma / 5
            Retornar this.promedio
        FinMetodo

        // Retorna el estado del estudiante segun su promedio
        ObtenerEstado(): Cadena
            Si this.promedio >= 10.5 entonces
                Retornar "APROBADO"
            Sino
                Retornar "DESAPROBADO"
            FinSi
        FinMetodo

FIN CLASE



PROGRAMA PRINCIPAL


INICIO
    Definir estudiantes[100] Como Estudiante
    Definir total            Como Entero
    Definir opcion           Como Entero
    total ← 0

    REPETIR

        Escribir ""
        Escribir "         MENU           "
        Escribir ""
        Escribir "1. Registrar estudiante"
        Escribir "2. Calcular promedios"

        Escribir "Seleccione una opcion:"
        Leer opcion

        ------------------------------
        CASO 1: REGISTRAR ESTUDIANTE
       ------------------------------

            Si total >= 100 entonces

            Sino
                Definir nom          Como Cadena
                Definir id           Como Cadena
                Definir notasTemp[5] Como Real

                Leer nom
                // Convertir a mayusculas para uniformidad
                nom ← ConvertirMayusculas(nom)

                Leer id

                id ← ConvertirMayusculas(id)

                Definir existe Como Booleano
                existe ← falso
                Para i ← 0 hasta total-1 hacer
                    Si estudiantes[i].idEstudiante = id entonces
                        existe ← verdadero
                    FinSi
                FinPara

                Si existe entonces
                    Escribir "ERROR: Ya existe un estudiante con ese ID."

                    Escribir "Ingrese las 5 notas (0 a 20):"
                    Para i ← 0 hasta 4 hacer

                            Leer nota

                                Escribir "  ERROR: La nota debe estar entre 0 y 20."
                            FinSi
                        HASTA QUE nota >= 0 Y nota <= 20
                        notasTemp[i] ← nota
                    FinPara

                    estudiantes[total] ← nuevo Estudiante(nom, id, notasTemp)

            Sino

                Definir orden  Como Entero

                Escribir "  3. Sin ordenar"
                Leer orden

                Definir j    Como Entero
                Definir temp Como Estudiante

                Si orden = 1 entonces
                    // Ordenar por promedio de mayor a menor
                    Para i ← 0 hasta total-2 hacer
                        Para j ← 0 hasta total-2-i hacer
                            Si estudiantes[j].promedio < estudiantes[j+1].promedio entonces
                                temp             ← estudiantes[j]
                                estudiantes[j]   ← estudiantes[j+1]
                                estudiantes[j+1] ← temp
                            FinSi
                        FinPara
                    FinPara

                SiNo Si orden = 2 entonces

                    Para i ← 0 hasta total-2 hacer
                        Para j ← 0 hasta total-2-i hacer

                                temp             ← estudiantes[j]
                                estudiantes[j]   ← estudiantes[j+1]
                                estudiantes[j+1] ← temp

                        FinPara
                    FinPara
                FinSi

                // Mostrar resultados
                Escribir "-------------------------------------------------"
                Escribir "  NOMBRE         | ID       | PROMEDIO | ESTADO"
                Escribir "-------------------------------------------------"

                    Escribir estudiantes[i].nombre,
                            " | ", estudiantes[i].idEstudiante,
                            " | ", estudiantes[i].promedio,
                            " | ", estudiantes[i].ObtenerEstado()
                FinPara
                Escribir "------------------------------------------"
            FinSi

                Definir encontrado       Como Booleano

                encontrado ← falso

                Escribir "  1. Por nombre"
                Escribir "  2. Por ID"

                textoBusqueda ← ConvertirMayusculas(textoBusqueda)

                Escribir ""
                Escribir "  RESULTADOS DE BUSQUEDA"
                Escribir ""

                Para i ← 0 hasta total-1 hacer

                    Si criterioBusqueda = 1 entonces

                        // Ejemplo: buscar "GABRIEL" encuentra "GABRIEL TORRES"

                            Escribir "ID       : ", estudiantes[i].idEstudiante
                            Escribir "Promedio : ", estudiantes[i].promedio
                            Escribir "Estado   : ", estudiantes[i].ObtenerEstado()

                            Escribir "ID       : ", estudiantes[i].idEstudiante
                            Escribir "Promedio : ", estudiantes[i].promedio
                            Escribir "Estado   : ", estudiantes[i].ObtenerEstado()
                            Escribir "------------------------------------"
                            encontrado ← verdadero
                        FinSi
                    FinSi

                FinPara

                Si NOT encontrado entonces

                FinSi

        CASO 4: LISTAR ESTUDIANTES
        ---------------------

            Si total = 0 entonces

                Definir i     Como Entero
                Definir j     Como Entero

                Escribir "  2. Por promedio (mayor a menor)"
                Escribir "  3. Sin ordenar"

                        Para j ← 0 hasta total-2-i hacer
                            Si estudiantes[j].nombre > estudiantes[j+1].nombre entonces
                                temp             ← estudiantes[j]
                                estudiantes[j]   ← estudiantes[j+1]
                                estudiantes[j+1] ← temp
                            FinSi
                        FinPara
                    FinPara

                SiNo Si orden = 2 entonces
                    Para i ← 0 hasta total-2 hacer
                        Para j ← 0 hasta total-2-i hacer
                            Si estudiantes[j].promedio < estudiantes[j+1].promedio entonces
                                temp             ← estudiantes[j]
                                estudiantes[j]   ← estudiantes[j+1]
                                estudiantes[j+1] ← temp

                Escribir ""

                             estudiantes[i].nombre, " | ",
                             estudiantes[i].idEstudiante, " | ",
